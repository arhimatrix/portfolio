def orbital_assembly(modules, fleet):
  sequence = plan_assembly_order(modules)
  for module in sequence:
    robot = assign_robot(fleet, module)
    robot.dock(module,
        tolerance=0.002)
    verify_structural_lock(module)
    update_digital_twin(module.state)
